class Solution {

    private final int[][] moves = {
        {1, 0},
        {-1, 0},
        {0, 1},
        {0, -1}
    };

    private int rows, cols;

    private int solve(int[][] matrix, int r, int c, int[][] memo) {

        if (memo[r][c] > 0) {
            return memo[r][c];
        }

        int best = 1;

        for (int i = 0; i < moves.length; i++) {

            int nextRow = r + moves[i][0];
            int nextCol = c + moves[i][1];

            if (nextRow < 0 || nextRow >= rows ||
                nextCol < 0 || nextCol >= cols) {
                continue;
            }

            if (matrix[nextRow][nextCol] > matrix[r][c]) {
                best = Math.max(
                    best,
                    solve(matrix, nextRow, nextCol, memo) + 1
                );
            }
        }

        memo[r][c] = best;
        return best;
    }

    public int longestIncreasingPath(int[][] matrix) {

        rows = matrix.length;
        cols = matrix[0].length;

        int[][] memo = new int[rows][cols];
        int longest = 0;

        for (int r = 0; r < rows; r++) {
            for (int c = 0; c < cols; c++) {
                longest = Math.max(
                    longest,
                    solve(matrix, r, c, memo)
                );
            }
        }

        return longest;
    }
}
